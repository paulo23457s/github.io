<script>
// Countdown Timer
function startFitnessTimer(duration, display) {
    let timer = duration, minutes, seconds;
    const interval = setInterval(function() {
        minutes = parseInt(timer / 60, 10);
        seconds = parseInt(timer % 60, 10);
    
        minutes = minutes < 10 ? "0" + minutes : minutes;
        seconds = seconds < 10 ? "0" + seconds : seconds;
    
        display.textContent = minutes + ":" + seconds;
    
        if (--timer < 0) {
            clearInterval(interval);
            document.getElementById('fitness-timer').innerHTML = '<span style="color:white">OFFER EXPIRED!</span> <a href="#" style="color:#f1c40f;text-decoration:underline">Click here if you need more time</a>';
        }
    }, 1000);
}

// FAQ Toggle
function toggleFitnessFAQ() {
    const questions = document.querySelectorAll('.fitness-faq-question');
    questions.forEach(question => {
        question.addEventListener('click', () => {
            const answer = question.nextElementSibling;
            answer.style.display = answer.style.display === 'block' ? 'none' : 'block';
        });
    });
}

// Initialize when page loads
document.addEventListener('DOMContentLoaded', function() {
    // Start 15-minute countdown
    startFitnessTimer(60 * 15, document.querySelector('#fitness-timer'));
    
    // Setup FAQ toggles
    toggleFitnessFAQ();
});
</script>
